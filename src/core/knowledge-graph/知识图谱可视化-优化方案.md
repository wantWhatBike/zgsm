# 知识图谱可视化功能优化方案

## 1. 需求背景

本项目是一个VSCode插件，需要实现知识图谱可视化功能。核心逻辑是：主webview启动完成后，用户通过设置tab页，点击知识图谱面板的可视化按钮，在VSCode编辑区打开一个新的、独立的webview panel，展示知识图谱关系图。

## 2. 现有代码结构分析

### 2.1 后端结构

- `src/core/knowledge-graph/KnowledgeGraphManager.ts`: 知识图谱管理核心类
- `src/core/knowledge-graph/core/GraphRetriever.ts`: 负责从知识图谱中检索数据
- `src/core/knowledge-graph/knowledgeGraphMessageHandler.ts`: 处理知识图谱相关消息
- `src/core/knowledge-graph/types.ts`: 知识图谱相关类型定义
- `packages/types/src/knowledge-graph.ts`: 共享的知识图谱类型定义

### 2.2 前端结构

- `webview-ui/src/components/settings/ZgsmKnowledgeGraphSettings.tsx`: 知识图谱设置面板，包含可视化按钮位置

```typescript
// 在GraphRetriever类中添加方法
public async getGraphData(options?: {
    maxNodes?: number;
    depth?: number;
    filterTypes?: string[];
}): Promise<GraphData> {
    // 实现从知识图谱存储中获取数据并转换为GraphData格式
    // 可以设置默认参数以控制返回数据量
    const maxNodes = options?.maxNodes || 5000;
    const depth = options?.depth || 2;
    
    // 从存储中检索节点和关系
    const nodes = await this.storageManager.getTopNodes(maxNodes);
    const links = await this.storageManager.getLinksForNodes(nodes.map(n => n.id));
    
    // 转换为前端需要的格式
    return {
        nodes: nodes.map(n => ({
            id: n.id,
            label: n.name,
            type: n.type,
            size: n.size,
            metadata: n.metadata
        })),
        links: links.map(l => ({
            source: l.sourceId,
            target: l.targetId,
            type: l.relationType,
            metadata: l.metadata
        })),
        metadata: {
            totalNodes: nodes.length,
            totalLinks: links.length,
            timestamp: new Date().toISOString()
        }
    };
}
```

在 `src/core/knowledge-graph/knowledgeGraphMessageHandler.ts` 中添加处理可视化相关消息的逻辑：

```typescript
import { GraphData, KNOWLEDGE_GRAPH_MESSAGES } from '@roo-code/types';

// 在类的构造函数中注册新的消息处理器
this.messageHandlers = {
    // 现有处理器...
    [KNOWLEDGE_GRAPH_MESSAGES.OPEN_VISUALIZATION]: this.handleOpenVisualization.bind(this),
    [KNOWLEDGE_GRAPH_MESSAGES.GET_GRAPH_DATA]: this.handleGetGraphData.bind(this),
};

// 添加处理打开可视化面板的方法
private async handleOpenVisualization(message: any): Promise<void> {
    try {
        // 检查知识图谱是否已构建完成
        const status = await this.knowledgeGraphManager.getStatus();
        
        if (status.status !== 'completed') {
            this.sendErrorResponse(message, 'Knowledge graph not built yet');
            return;
        }
        
        // 创建并显示新的webview面板
        await this.createVisualizationPanel();
        
        this.sendSuccessResponse(message);
    } catch (error) {
        this.sendErrorResponse(message, error instanceof Error ? error.message : String(error));
    }
}

// 添加获取图谱数据的方法
private async handleGetGraphData(message: any): Promise<void> {
    try {
        const options = message.payload || {};
        const graphData = await this.knowledgeGraphManager.getGraphData(options);
        
        this.sendResponse(message, KNOWLEDGE_GRAPH_MESSAGES.GRAPH_DATA_RESPONSE, { 
            graphData 
        });
    } catch (error) {
        this.sendErrorResponse(message, error instanceof Error ? error.message : String(error));
    }
}

// 添加创建可视化面板的方法
private async createVisualizationPanel(): Promise<void> {
    // 实现创建新的VSCode Webview Panel的逻辑
    // 使用vscode.window.createWebviewPanel创建独立的面板
    const panel = vscode.window.createWebviewPanel(
        'knowledgeGraphVisualization',
        'Knowledge Graph Visualization',
        vscode.ViewColumn.One,
        {
            enableScripts: true,
            retainContextWhenHidden: true
        }
    );
    
    // 设置HTML内容，加载前端可视化组件
    panel.webview.html = this.getVisualizationHtml(panel.webview);
    
    // 设置消息监听
    panel.webview.onDidReceiveMessage(
        message => this.handleVisualizationMessage(message, panel),
        undefined,
        this.disposables
    );
}

// 添加获取可视化HTML的方法
private getVisualizationHtml(webview: vscode.Webview): string {
    // 生成加载前端可视化组件的HTML
    const nonce = this.getNonce();
    const scriptUri = webview.asWebviewUri(vscode.Uri.joinPath(
        this.context.extensionUri,
        'out',
        'webview-ui',
        'knowledgeGraphVisualization.js'
    ));
    
    return `
        <!DOCTYPE html>
        <html lang="en">
        <head>
            <meta charset="UTF-8">
            <meta name="viewport" content="width=device-width, initial-scale=1.0">
            <title>Knowledge Graph Visualization</title>
            <meta http-equiv="Content-Security-Policy" content="default-src 'none'; script-src 'nonce-${nonce}';">
        </head>
        <body>
            <div id="root"></div>
            <script nonce="${nonce}" src="${scriptUri}"></script>
        </body>
        </html>
    `;
}

// 添加处理可视化面板消息的方法
private handleVisualizationMessage(message: any, panel: vscode.WebviewPanel): void {
    switch (message.type) {
        case KNOWLEDGE_GRAPH_MESSAGES.GET_GRAPH_DATA:
            this.handleGetGraphData(message);
            break;
        default:
            console.warn(`Unknown message type: ${message.type}`);
    }
}
```

#### 3.2.2 创建知识图谱可视化组件

在 `webview-ui/src/components/knowledge-graph` 目录下创建以下文件：

1. `ForceGraph.tsx`: D3力导向图组件

```typescript
import React, { useRef, useEffect, useState } from 'react';
import * as d3 from 'd3';
import { GraphData } from '@roo-code/types';

interface ForceGraphProps {
    graphData: GraphData;
    width?: number;
    height?: number;
}

export const ForceGraph: React.FC<ForceGraphProps> = ({ 
    graphData, 
    width = 1200, 
    height = 800 
}) => {
    const svgRef = useRef<SVGSVGElement>(null);
    const [isLoading, setIsLoading] = useState(false);

    useEffect(() => {
        if (!svgRef.current || !graphData) return;
        
        setIsLoading(true);
        
        // 清理之前的内容
        d3.select(svgRef.current).selectAll('*').remove();
        
        // 创建SVG元素
        const svg = d3.select(svgRef.current)
            .attr('width', width)
            .attr('height', height);
        
        // 创建缩放行为
        const zoom = d3.zoom()
            .scaleExtent([0.1, 4])
            .on('zoom', (event) => {
                g.attr('transform', event.transform);
            });
        
        svg.call(zoom as any);
        
        // 创建包含所有图形元素的g元素
        const g = svg.append('g');
        
        // 创建力模拟
        const simulation = d3.forceSimulation(graphData.nodes)
            .force('link', d3.forceLink(graphData.links).id((d: any) => d.id).distance(100))
            .force('charge', d3.forceManyBody().strength(-300))
            .force('center', d3.forceCenter(width / 2, height / 2))
            .force('collision', d3.forceCollide().radius(50));
        
        // 绘制连接线
        const link = g.append('g')
            .attr('stroke', '#999')
            .attr('stroke-opacity', 0.6)
            .selectAll('line')
            .data(graphData.links)
            .join('line')
            .attr('stroke-width', 2);
        
        // 创建节点组
        const node = g.append('g')
            .attr('stroke', '#fff')
            .attr('stroke-width', 1.5)
            .selectAll('.node')
            .data(graphData.nodes)
            .join('g')
            .attr('class', 'node')
            .call(drag(simulation));
        
        // 添加节点圆圈
        node.append('circle')
            .attr('r', 8)
            .attr('fill', (d: any) => {
                // 根据节点类型设置不同颜色
                switch (d.type) {
                    case 'file': return '#3b82f6';
                    case 'directory': return '#10b981';
                    case 'module': return '#8b5cf6';
                    default: return '#6b7280';
                }
            });
        
        // 添加节点标签
        node.append('text')
            .attr('dx', 12)
            .attr('dy', '.35em')
            .text((d: any) => d.label)
            .attr('fill', 'white')
            .attr('font-size', 12);
        
        // 双击节点打开文件
        node.on('dblclick', (event, d: any) => {
            vscode.postMessage({ type: 'openFile', payload: { filePath: d.id } });
        });
        
        // 定义拖拽行为
        function drag(simulation: any) {
            function dragstarted(event: any, d: any) {
                if (!event.active) simulation.alphaTarget(0.3).restart();
                d.fx = d.x;
                d.fy = d.y;
            }
            
            function dragged(event: any, d: any) {
                d.fx = event.x;
                d.fy = event.y;
            }
            
            function dragended(event: any, d: any) {
                if (!event.active) simulation.alphaTarget(0);
                d.fx = null;
                d.fy = null;
            }
            
            return d3.drag()
                .on('start', dragstarted)
                .on('drag', dragged)
                .on('end', dragended);
        }
        
        // 更新力模拟
        simulation.on('tick', () => {
            link
                .attr('x1', (d: any) => d.source.x)
                .attr('y1', (d: any) => d.source.y)
                .attr('x2', (d: any) => d.target.x)
                .attr('y2', (d: any) => d.target.y);
            
            node
                .attr('transform', (d: any) => `translate(${d.x},${d.y})`);
        });
        
        setIsLoading(false);
        
        // 清理函数
        return () => {
            simulation.stop();
        };
    }, [graphData, width, height]);
    
    return (
        <div className="relative">
            {isLoading && (
                <div className="absolute inset-0 flex items-center justify-center bg-black bg-opacity-50">
                    <div className="text-white">Loading graph...</div>
                </div>
            )}
            <svg ref={svgRef} className="bg-[#050505] w-full h-full" />
        </div>
    );
};
```

2. `GraphVisualizer.tsx`: 知识图谱可视化主组件

```typescript
import React, { useEffect, useState, useCallback } from 'react';
import { ForceGraph } from './ForceGraph';
import { GraphData, KNOWLEDGE_GRAPH_MESSAGES } from '@roo-code/types';
import { vscode } from '@/utils/vscode';

// 确保 window.vscode 存在
const vscode = window.vscode || { postMessage: () => {} };

export const GraphVisualizer: React.FC = () => {
    const [graphData, setGraphData] = useState<GraphData | null>(null);
    const [isLoading, setIsLoading] = useState(true);
    const [error, setError] = useState<string | null>(null);
    
    // 加载知识图谱数据
    const loadGraphData = useCallback(() => {
        setIsLoading(true);
        setError(null);
        
        // 发送消息获取图谱数据
        vscode.postMessage({
            type: KNOWLEDGE_GRAPH_MESSAGES.GET_GRAPH_DATA,
            payload: {
                maxNodes: 5000, // 控制初始加载的节点数量
                depth: 2
            }
        });
    }, []);
    
    // 设置消息监听器
    useEffect(() => {
        const handleMessage = (event: MessageEvent) => {
            const message = event.data;
            
            if (message.type === KNOWLEDGE_GRAPH_MESSAGES.GRAPH_DATA_RESPONSE) {
                if (message.error) {
                    setError(message.error);
                } else if (message.payload?.graphData) {
                    setGraphData(message.payload.graphData);
                }
                setIsLoading(false);
            }
        };
        
        window.addEventListener('message', handleMessage);
        
        // 初始加载数据
        loadGraphData();
        
        return () => {
            window.removeEventListener('message', handleMessage);
        };
    }, [loadGraphData]);
    
    // 渲染错误状态
    if (error) {
        return (
            <div className="flex flex-col items-center justify-center h-screen p-4 text-white">
                <h2 className="text-xl font-bold mb-2">Error</h2>
                <p className="mb-4">{error}</p>
                <button 
                    onClick={loadGraphData} 
                    className="px-4 py-2 bg-blue-600 rounded hover:bg-blue-700"
                >
                    Retry
                </button>
            </div>
        );
    }
    
    // 渲染加载状态
    if (isLoading) {
        return (
            <div className="flex flex-col items-center justify-center h-screen text-white">
                <div className="animate-spin rounded-full h-12 w-12 border-t-2 border-b-2 border-blue-500 mb-4"></div>
                <p>Loading knowledge graph...</p>
            </div>
        );
    }
    
    // 渲染知识图谱
    return (
        <div className="h-screen w-screen bg-[#050505]">
            {graphData && (
                <ForceGraph 
                    graphData={graphData} 
                    width={window.innerWidth} 
                    height={window.innerHeight} 
                />
            )}
        </div>
    );
};
```
